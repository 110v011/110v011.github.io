<script type="module">
import * as THREE from "./three.module.js";  // ★ローカル読み込み

// ===== Scene =====
const scene = new THREE.Scene();
scene.fog = new THREE.Fog(0xcccc55, 5, 40);

// ===== Camera =====
const camera = new THREE.PerspectiveCamera(75, innerWidth/innerHeight, 0.1, 1000);
camera.position.set(0, 1.7, 0);

// ===== Renderer =====
const renderer = new THREE.WebGLRenderer({antialias:true});
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

// ===== Lights =====
scene.add(new THREE.AmbientLight(0xffffcc, 0.4));
const lamp = new THREE.PointLight(0xffffaa, 1.5, 30);
lamp.position.set(0, 3, 0);
scene.add(lamp);

// ===== Floor & Ceiling =====
const floorMat = new THREE.MeshPhongMaterial({color:0xdddd88});
const ceilMat  = new THREE.MeshPhongMaterial({color:0xffffdd});
const planeGeo = new THREE.PlaneGeometry(200, 200);

const floor = new THREE.Mesh(planeGeo, floorMat);
floor.rotation.x = -Math.PI/2;
scene.add(floor);

const ceil = new THREE.Mesh(planeGeo, ceilMat);
ceil.rotation.x = Math.PI/2;
ceil.position.y = 3;
scene.add(ceil);

// ===== Walls =====
const wallMat = new THREE.MeshPhongMaterial({color:0xeeee66});
const wallGeo = new THREE.BoxGeometry(1, 3, 10);
const colliders = [];

function wall(x, z, rot=0){
  const w = new THREE.Mesh(wallGeo, wallMat);
  w.position.set(x, 1.5, z);
  w.rotation.y = rot;
  scene.add(w);
  colliders.push(w);
}

// Simple maze layout
wall(0,-5); wall(5,0,Math.PI/2); wall(-5,0,Math.PI/2);
wall(0,5); wall(10,0,Math.PI/2); wall(-10,0,Math.PI/2);
wall(0,-15); wall(0,15); wall(15,0,Math.PI/2); wall(-15,0,Math.PI/2);

// ===== Input =====
const keys = {w:0,a:0,s:0,d:0};
onkeydown = e => keys[e.key] = 1;
onkeyup   = e => keys[e.key] = 0;

// ===== Mouse Look =====
let yaw=0, pitch=0, md=false, lx=0, ly=0;
onmousedown = e => { md=true; lx=e.clientX; ly=e.clientY; };
onmouseup   = () => md=false;
onmousemove = e => {
  if(!md) return;
  yaw -= (e.clientX - lx)*0.002;
  pitch -= (e.clientY - ly)*0.002;
  pitch = Math.max(-1.3, Math.min(1.3, pitch));
  lx=e.clientX; ly=e.clientY;
};

// ===== Collision =====
function hit(pos){
  const r=0.4;
  for(const w of colliders){
    const box = new THREE.Box3().setFromObject(w).expandByScalar(r);
    if(box.containsPoint(pos)) return true;
  }
  return false;
}

// ===== Loop =====
function loop(){
  requestAnimationFrame(loop);

  const forward = new THREE.Vector3(Math.sin(yaw),0,Math.cos(yaw));
  const right   = new THREE.Vector3().crossVectors(forward,new THREE.Vector3(0,1,0)).negate();
  let move = new THREE.Vector3();

  if(keys.w) move.add(forward);
  if(keys.s) move.sub(forward);
  if(keys.a) move.sub(right);
  if(keys.d) move.add(right);

  if(move.length()>0){
    move.normalize().multiplyScalar(0.08);
    const np = camera.position.clone().add(move);
    if(!hit(np)) camera.position.copy(np);
  }

  const dir = new THREE.Vector3(
    Math.sin(yaw)*Math.cos(pitch),
    Math.sin(pitch),
    Math.cos(yaw)*Math.cos(pitch)
  );
  camera.lookAt(camera.position.clone().add(dir));

  renderer.render(scene, camera);
}
loop();
</script>
